# INSTALLATION OF MPF V0.57.0 ON A LINUX MACHINE

# *******************************************
# TERMINAL COMMANDS :: COPY AND PASTE THESE INTO A TERMINAL WINDOW
# *******************************************

# ------------------------------
# UPDATE APT-GET
# ------------------------------
sudo apt-get update

# ------------------------------
# INSTALL GRAPHICS DEPENDENCIES
# ------------------------------
sudo apt-get install -y \
  build-essential git curl \
  libsqlite3-dev zlib1g-dev \
  libjpeg-dev libtiff5-dev \
  libsdl2-dev libsdl2-ttf-dev libsdl2-image-dev libsdl2-mixer-dev \
  gstreamer1.0-plugins-base gstreamer1.0-plugins-bad gstreamer1.0-plugins-ugly \
  libgstreamer1.0-dev \
  libswscale-dev libavformat-dev libavcodec-dev libavfilter-dev libavdevice-dev \
  libgl1-mesa-dev libgles2-mesa-dev libgl1-mesa-dri \
  pulseaudio libportaudio2 dbus-x11 libasound-dev \
  libunwind-dev lsb-release

# ------------------------------
# INSTALL PYENV
# ------------------------------
curl https://pyenv.run | bash

# ------------------------------
# ADD PYENV TO SHELL CONFIG
# ------------------------------
echo 'export PYENV_ROOT="$HOME/.pyenv"' >> ~/.bashrc
echo '[[ -d $PYENV_ROOT/bin ]] && export PATH="$PYENV_ROOT/bin:$PATH"' >> ~/.bashrc
echo 'eval "$(pyenv init -)"' >> ~/.bashrc
source ~/.bashrc

# ------------------------------
# INSTALL PYTHON AND MAKE THIS VERSION GLOBAL
# ------------------------------
pyenv install 3.11.8
pyenv global 3.11.8


# ------------------------------
# CREATE A CLEAN PYTHON ENVIRONMENT
# The name "Py311_MPF057" is arbitrary, but keep it consistant
# ------------------------------
pyenv shell 3.11.8
python -m venv ~/venvs/Py311_MPF057
source ~/venvs/Py311_MPF057/bin/activate


# ------------------------------
# INSTALL DEPENDENCIES SPECIFIC TO V0.57.0
# ------------------------------
# This step need to be done while in the named venv created above
# ------------------------------

# SETUP TOOLS
pip install 'setuptools==69.1.0'

# CYTHON
pip install Cython==0.29.36

# RUAML.YAML
pip install 'ruamel.yaml==0.18.6'

# RUAML.YAML.CLIB
pip install 'ruamel.yaml.clib==0.2.8'

# PILLOW
pip install pillow==9.5.0

# STABLE VERSION OF MPF V0.57.0
pip install mpf==0.57.0.dev41

# STABLE VERSION OF MPF-MC
pip install mpf-mc==0.57.0.dev14

# ------------------------------
# VERIFY INSTALL
# ------------------------------
mpf --version

# ------------------------------
# ...OR
# ------------------------------
pip list


# ------------------------------
# CREATE A SHELL SCRIPT TO RUN THE GAME
# Copy the text below into a file called "Linux_BAM_GITHUB_v57_Both.sh"
# ------------------------------

#!/bin/bash

#ACTIVATE VIRTUAL ENVIRONMENT
source ~/venvs/Py311_MPF057/bin/activate

# SET MAIN GAME DIRECTORY
cd Dropbox/GitHub/MTWABP/FunAndGames/PINBALL/MPF_GAMES_v57/

# GOTO SPECIFIC GAME FOLDER
cd BAM_TOTEM_v57/

sleep 2s
mpf both -xvVaA


# ------------------------------
# ADD AUTO-LAUNCH ON POWERUP
# ------------------------------

1) Goto Menu > Search Bar > Startup Applications

2) Click "+" > Custom Command

3) Add Startup Program
    Name > LaunchMPFv57
    Command > gnome-terminal --title="MPFv57" -- bash -c "cd ~ && ./Linux_BAM_GITHUB_v57_Both.sh; exec bash"
    Comment > none
    Startup Delay > 2
    Click "Add"

4) Test the function of the startup command
    Highlight the newly created command
    Click the ">" play button


# ------------------------------
# INSTALL "MONITOR" IN SEPARATE PYTHON ENVIRONMENT
# The name "Py311_MPF057_monitor" is arbitrary, but keep it consistant
# ------------------------------

# CREATE NEW CLEAN ENVIRONMENT
python -m venv ~/venvs/Py311_MPF057_monitor
source ~/venvs/Py311_MPF057_monitor/bin/activate

# INSTALL ARBITRARY MPF VERSION

For a normal Linux install, the relevant parts are basically:

sudo apt-get update

sudo apt-get install -y \
  build-essential git curl \
  libsqlite3-dev zlib1g-dev \
  libjpeg-dev libtiff5-dev \
  libsdl2-dev libsdl2-ttf-dev libsdl2-image-dev libsdl2-mixer-dev \
  gstreamer1.0-plugins-base gstreamer1.0-plugins-bad gstreamer1.0-plugins-ugly \
  libgstreamer1.0-dev \
  libswscale-dev libavformat-dev libavcodec-dev libavfilter-dev libavdevice-dev \
  libgl1-mesa-dev libgles2-mesa-dev libgl1-mesa-dri \
  pulseaudio libportaudio2 dbus-x11 libasound-dev \
  libunwind-dev lsb-release

Then install Python 3.10.15 using pyenv:

curl https://pyenv.run | bash

Add PyEnv to your shell config:

echo 'export PYENV_ROOT="$HOME/.pyenv"' >> ~/.bashrc
echo '[[ -d $PYENV_ROOT/bin ]] && export PATH="$PYENV_ROOT/bin:$PATH"' >> ~/.bashrc
echo 'eval "$(pyenv init -)"' >> ~/.bashrc
source ~/.bashrc

Install and select Python:

pyenv install 3.10.15
pyenv global 3.10.15

Then create a clean virtual environment:

python -m venv ~/venvs/mpf57
source ~/venvs/mpf57/bin/activate

Install the pinned/stable package versions:

pip install --upgrade pip 'setuptools<73' wheel build
pip install 'pillow<10' 'packaging<24'
pip install Cython==0.29.36
pip install mpf==0.57.4
pip install mpf-mc==0.57.1

Verify:

mpf --version
mpf both --version
